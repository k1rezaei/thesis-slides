\newcommand{\roadmaptarget}[1]{\hypertarget{#1}{}}
\newcommand{\rmstation}[4]{%
    % #1 = x, #2 = index, #3 = link target, #4 = label
    \ifnum\rmcur=#2
        \draw[themecolor!18,line width=0pt,fill=themecolor!12] (#1,0) circle (0.78);
        \node[circle,fill=themecolor,draw=white,line width=1.6pt,text=white,
            font=\bfseries\large,minimum size=1.05cm,inner sep=0pt] at (#1,0) {#2};
        \node[anchor=north,align=center,text width=4.1cm,font=\small\bfseries,
            text=themecolor] at (#1,-1.05)
        {\hypersetup{linkcolor=themecolor}\hyperlink{#3}{#4}};
    \else
        \node[circle,fill=white,draw=gray!45,line width=1pt,text=gray!55,
            font=\bfseries,minimum size=0.78cm,inner sep=0pt] at (#1,0) {#2};
        \node[anchor=north,align=center,text width=4.1cm,font=\small,
            text=gray!45!black] at (#1,-0.92)
        {\hypersetup{linkcolor=gray!45!black}\hyperlink{#3}{#4}};
    \fi
}
