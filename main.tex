\begin{frame}{\textsc{LocoGen}: Searching for the Controlling Layers}
    \centering
    \uncover<1->{%
        Which $m$ of the $M$ cross-attention layers should get $c'$?\\
        \textbf{Brute force} tries all $\binom{M}{m}$ subsets
        {\scriptsize\textcolor{xattn}{(SD-XL: 12M)}}.}

    \vspace{0.5em}
    \uncover<2->{%
        Restricting the search to \textcolor{themecolor}{\textbf{adjacent}} layers
        leaves only $M-m+1$ windows
        {\scriptsize\textcolor{themecolor}{(SD-XL: 66)}}.}

    \vspace{0.7em}
    \uncover<3->{%
        \begin{tcolorbox}[colback=themecolor!4,colframe=themecolor,arc=1.5mm,boxrule=0.8pt,
                width=0.92\textwidth,left=6pt,right=6pt,top=4pt,bottom=4pt,
                fonttitle=\bfseries\small,title=\textsc{LocoGen}]
            \small
            \textbf{for} $j = 1, \dots, M-m+1$:\\
            \hspace*{1.3em}$C' \gets \{C_j, \dots, C_{j+m-1}\}$
            \hfill{\scriptsize\color{gray!45!black} a window of $m$ adjacent layers}\\
            \hspace*{1.3em}generate $N$ images: layers in $C'$ receive $c'_i$, all others receive $c_i$\\
            \hspace*{1.3em}$a_j \gets \operatorname{avg}_i \textsc{CLIP-Score}(T_i, I_\text{altered})$\\
            \textbf{return} $\arg\min_j a_j$
            \hfill{\scriptsize\color{gray!45!black} style, objects\quad($\arg\max$ for facts)}
        \end{tcolorbox}}

    \vspace{0.2em}
    \uncover<4->{\small
        $m$ itself is swept upward: the \textbf{smallest} window that removes the attribute.}
\end{frame}

\begin{frame}{\textsc{LocoEdit}: Rewrite the Located Layers}
    \centering
    \uncover<1->{%
        \textbf{$N$ prompt pairs} drawn from the whole attribute.
        \vspace{0.4em}
        \begin{columns}[T,onlytextwidth]
            \column{0.48\textwidth}
            \begin{tcolorbox}[colback=xattn!6,colframe=xattn,arc=1.5mm,boxrule=0.8pt,
                    left=4pt,right=4pt,top=3pt,bottom=3pt,
                    fonttitle=\bfseries\small,coltitle=white,colbacktitle=xattn,
                    title=$C_\text{orig}$]
                \footnotesize
                \textit{``A house in the style of Van Gogh''}\\
                \textit{``A starry night in the style of Van Gogh''}\\
                $\vdots$
            \end{tcolorbox}
            \column{0.48\textwidth}
            \begin{tcolorbox}[colback=themecolor!6,colframe=themecolor,arc=1.5mm,boxrule=0.8pt,
                    left=4pt,right=4pt,top=3pt,bottom=3pt,
                    fonttitle=\bfseries\small,coltitle=white,colbacktitle=themecolor,
                    title=$C_\text{target}$]
                \footnotesize
                \textit{``a painting of a house''}\\
                \textit{``a painting of a starry night''}\\
                $\vdots$
            \end{tcolorbox}
        \end{columns}}

    \vspace{0.1em}
    \uncover<2->{%
        \vspace{-0.4em}
        \[
            \min_{W^K}\ \big\| C_\text{orig} W^K - C_\text{target} \hat{W}^K \big\|_2^2
            + \lambda_K \big\| W^K - \hat{W}^K \big\|_2^2
        \]}
    \vspace{-0.7em}
    \uncover<3->{%
        \[
            W^K = \big(C_\text{orig}^{\top} C_\text{orig} + \lambda_K I\big)^{-1}
            \big(C_\text{orig}^{\top} C_\text{target}\hat{W}^K + \lambda_K \hat{W}^K\big)
        \]
        \vspace{-1.1em}
        \begin{center}
            {\small one linear solve: no gradients, no fine-tuning; same for $W^V$.}
        \end{center}}
\end{frame}

\begin{frame}{Localizing by Attention Contribution}
    \centering
    How hard does text token $\mathbf{x}_j$ \textbf{push} on image token $\mathbf{y}_i$
    in \textbf{this block}?

    \begin{minipage}[c][1.45cm][c]{\textwidth}
        \centering
        \only<1>{$\displaystyle
                \text{cont}_{i,j} \;=\; \big\| \text{attn}_{i,j}\; \mathbf{v}_j\, W_o \big\|_2 $}%
        \only<2->{$\displaystyle
                \text{cont}_{i,j} \;=\; \Big\| \sum_{h=1}^{H} \text{attn}^h_{i,j}\;
                \mathbf{v}^h_j\, W^h_o \Big\|_2 $}
    \end{minipage}

    \vspace{-0.3em}
    \uncover<2->{\small Attention runs $H$ heads in parallel, so add their writes first.}

    \vspace{0.3em}
    \begin{columns}[c,onlytextwidth]
        \column{0.64\textwidth}
        \uncover<3->{\small
            Summing over image tokens $i$ gives how much $\mathbf{x}_j$ writes into the
            \textbf{whole image} at this layer.}

        \vspace{0.5em}
        \uncover<4->{\small
            For a meaningful $\mathbf{x}_j$ (an artist, an object, a character),
            rank the layers by that score and keep the
            \textcolor{themecolor}{\textbf{top contributors}}.}

        \column{0.32\textwidth}
        \uncover<4->{\includegraphics[width=\linewidth]{Figures/dit/attribution-ranking.jpg}}
    \end{columns}

\begin{frame}{How Unlearning Is Done Today}
    \centering
    \vspace{0.6em}
    {\small\[
            \theta_\text{unlearn}=\arg\min_\theta\;
            \underbrace{\mathbb{E}_{x\sim\mathcal{D}_f}\big[\mathcal{L}_f(x;\theta)\big]}_{\text{\normalfont\color{xattn}forget}}
            \;+\;\lambda\,
            \underbrace{\mathbb{E}_{x\sim\mathcal{D}_r}\big[\mathcal{L}_r(x;\theta)\big]}_{\text{\normalfont\color{themecolor}retain}}
        \]}

    \vspace{0.2em}
    {\small \textcolor{xattn}{\textbf{Make the forget documents unpredictable.}}
        \textcolor{themecolor}{\textbf{Keep everything else.}}}

    \vspace{1.0em}
    \uncover<2->{%
        \begin{columns}[T,onlytextwidth]
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{Gradient ascent}\\[0.4em]
            $\mathcal{L}_\text{GA}=-\dfrac{1}{|x|}\sum_i \log P_\theta(x_i\mid x_{<i})$
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{KL divergence}\\[0.4em]
            $\mathcal{L}_\text{KL}=-\dfrac{1}{|x|}\sum_i \mathrm{KL}\big(P_\theta\,\|\,P_{\theta_\mathcal{D}}\big)$
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{NPO}\\[0.4em]
            $\mathcal{L}_\text{NPO}=\dfrac{2}{\beta|x|}\sum_i \log\!\Big(1+\big(\tfrac{P_\theta}{P_{\theta_\mathcal{D}}}\big)^{\!\beta}\Big)$
        \end{columns}}

    \vspace{1.0em}
    \uncover<3->{\small All three \textbf{push probability away} from the forget documents.}
\end{frame}

\begin{frame}{Maximizing a Loss Is a Blunt Instrument}
    \centering
    \vspace{0.7em}
    {\small Every method so far \textbf{pushes probability down} on $\mathcal{D}_f$.
        Where it lands is not controlled.}

    \vspace{1.0em}

        keep training $\theta_\mathcal{D}$ on $\mathcal{D}_f$, then step \textbf{backwards}.}

    \vspace{0.6em}
    \uncover<2->{\small\[
            \vec\theta_f=\theta'_\mathcal{D}-\theta_\mathcal{D},

            \theta_\text{unlearn}=\theta_\mathcal{D}-\alpha\,\vec\theta_f
        \]}

    \vspace{0.6em}
    \uncover<3->{\small But $\theta_\mathcal{D}$ has \textit{already} absorbed $\mathcal{D}_f$.
        The direction it gives back is \textcolor{xattn}{\textbf{weak}}.}

    \vspace{0.7em}
    \uncover<4->{\small \textcolor{themecolor}{\textbf{What if we took the direction from
                before the data arrived?}}}
\end{frame}

\begin{frame}{MSA: How We Ran It}

\begin{frame}{What the Two Papers Say Together}
    \centering
    \vspace{0.8em}
    \begin{itemize}\small
        \item Concept-level benchmarks reward \textbf{silence}. \RESTOR\ asks for \textbf{recovery}.
        \item Loss-maximizing methods forget, then drift to random.
        \item No single method wins on every metric.
        \item \textbf{Training history is a resource}: a checkpoint from before the data
              gives a cleaner estimate of what that data did.
    \end{itemize}
\end{frame}
