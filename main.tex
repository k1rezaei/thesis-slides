\begin{frame}{\textsc{LocoGen}: Searching for the Controlling Layers}
    \centering
    \uncover<1->{%
        Which $m$ of the $M$ cross-attention layers should get $c'$?\\
        \textbf{Brute force} tries all $\binom{M}{m}$ subsets.\\
        {\scriptsize\color{gray!45!black} SD-XL: $M=70$, $m=5$
            $\Rightarrow$ \textcolor{xattn}{\textbf{12M}} subsets\uncover<2->{,
                only \textcolor{themecolor}{\textbf{66}} windows}.}}

    \vspace{0.5em}
    \uncover<2->{%
        Restricting the search to \textcolor{themecolor}{\textbf{adjacent}} layers
        leaves only $M-m+1$ windows.}

    \vspace{0.7em}
    \uncover<3->{%
        \begin{tcolorbox}[colback=themecolor!4,colframe=themecolor,arc=1.5mm,boxrule=0.8pt,
                width=0.92\textwidth,left=6pt,right=6pt,top=4pt,bottom=4pt,
                fonttitle=\bfseries\small,title=\textsc{LocoGen}]
            \small
            \textbf{for} $j = 1, \dots, M-m+1$:\\
            \hspace*{1.3em}$C' \gets \{C_j, \dots, C_{j+m-1}\}$
            \hfill{\scriptsize\color{gray!45!black} a window of $m$ adjacent layers}\\
            \hspace*{1.3em}generate $N$ images: layers in $C'$ receive $c'_i$, all others receive $c_i$\\
            \hspace*{1.3em}$a_j \gets \operatorname{avg}_i \textsc{CLIP-Score}(T_i, I_\text{altered})$\\
            \textbf{return} $\arg\min_j a_j$
            \hfill{\scriptsize\color{gray!45!black} style, objects\quad($\arg\max$ for facts)}
        \end{tcolorbox}}

    \vspace{0.2em}
    \uncover<4->{\small
        $m$ itself is swept upward: the \textbf{smallest} window that removes the attribute.}
\end{frame}

\begin{frame}{\textsc{LocoEdit}: Rewrite the Located Layers}
    \centering
    \uncover<1->{%
        \textbf{$N$ prompt pairs} drawn from the whole attribute.
        \vspace{0.4em}
        \begin{columns}[T,onlytextwidth]
            \column{0.48\textwidth}
            \begin{tcolorbox}[colback=xattn!6,colframe=xattn,arc=1.5mm,boxrule=0.8pt,
                    left=4pt,right=4pt,top=3pt,bottom=3pt,
                    fonttitle=\bfseries\small,coltitle=white,colbacktitle=xattn,
                    title=$C_\text{orig}$]
                \footnotesize
                \textit{``A house in the style of Van Gogh''}\\
                \textit{``A starry night in the style of Van Gogh''}\\
                $\vdots$
            \end{tcolorbox}
            \column{0.48\textwidth}
            \begin{tcolorbox}[colback=themecolor!6,colframe=themecolor,arc=1.5mm,boxrule=0.8pt,
                    left=4pt,right=4pt,top=3pt,bottom=3pt,
                    fonttitle=\bfseries\small,coltitle=white,colbacktitle=themecolor,
                    title=$C_\text{target}$]
                \footnotesize
                \textit{``a painting of a house''}\\
                \textit{``a painting of a starry night''}\\
                $\vdots$
            \end{tcolorbox}
        \end{columns}}

    \vspace{0.1em}
    \uncover<2->{%
        \vspace{-0.4em}
        \[
            \min_{W^K}\ \big\| C_\text{orig} W^K - C_\text{target}
            \overset{\text{\tiny\color{gray!45!black}original}}{\hat{W}^K} \big\|_2^2
            + \lambda_K \big\| W^K - \hat{W}^K \big\|_2^2
        \]}
    \vspace{-0.7em}
    \uncover<3->{%
        \[
            W^K = \big(C_\text{orig}^{\top} C_\text{orig} + \lambda_K I\big)^{-1}
            \big(C_\text{orig}^{\top} C_\text{target}\hat{W}^K + \lambda_K \hat{W}^K\big)
        \]
        \vspace{-1.1em}
        \begin{center}
            {\small one linear solve: no gradients, no fine-tuning; same for $W^V$.}
        \end{center}}
\end{frame}

\begin{frame}{Localizing by Attention Contribution}
    \centering
    How hard does text token $\mathbf{x}_j$ \textbf{push} on image token $\mathbf{y}_i$
    in \textbf{this block}?

    \begin{minipage}[c][1.45cm][c]{\textwidth}
        \centering
        \only<1>{$\displaystyle
                \text{cont}_{i,j} \;=\; \big\| \text{attn}_{i,j}\; \mathbf{v}_j\, W_o \big\|_2 $}%
        \only<2->{$\displaystyle
                \text{cont}_{i,j} \;=\; \Big\| \sum_{h=1}^{H} \text{attn}^h_{i,j}\;
                \mathbf{v}^h_j\, W^h_o \Big\|_2 $}
    \end{minipage}

    \vspace{-0.3em}
    \uncover<2->{\small Attention runs $H$ heads in parallel, so add their writes first.}

    \vspace{0.3em}

        \column{0.64\textwidth}
        \uncover<3->{\small
            Summing over image tokens $i$ gives how much $\mathbf{x}_j$ writes into the
            \textbf{whole image} at this layer.}

        \vspace{0.5em}
        \uncover<4->{\small
            For a meaningful $\mathbf{x}_j$ (an artist, an object, a character),
            rank the layers by that score and keep the
            \textcolor{themecolor}{\textbf{top contributors}}.}

\begin{frame}{How Unlearning Is Done Today}
    \centering
    \vspace{0.6em}
    {\small\[
            \theta_\text{unlearn}=\arg\min_\theta\;
            \underbrace{\mathbb{E}_{x\sim\mathcal{D}_f}\big[\mathcal{L}_f(x;\theta)\big]}_{\text{\normalfont\color{xattn}forget}}
            \;+\;\lambda\,
            \underbrace{\mathbb{E}_{x\sim\mathcal{D}_r}\big[\mathcal{L}_r(x;\theta)\big]}_{\text{\normalfont\color{themecolor}retain}}
        \]}

    \vspace{0.2em}
    {\small \textcolor{xattn}{\textbf{Make the forget documents unpredictable.}}
        \textcolor{themecolor}{\textbf{Keep everything else.}}}

    \vspace{1.0em}
    \uncover<2->{%
        \begin{columns}[T,onlytextwidth]
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{Gradient ascent}\\[0.4em]
            $\mathcal{L}_\text{GA}=-\dfrac{1}{|x|}\sum_i \log P_\theta(x_i\mid x_{<i})$
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{KL divergence}\\[0.4em]
            $\mathcal{L}_\text{KL}=-\dfrac{1}{|x|}\sum_i \mathrm{KL}\big(P_\theta\,\|\,P_{\theta_\mathcal{D}}\big)$
            \column{0.31\textwidth}
            \centering\scriptsize
            \textbf{NPO}\\[0.4em]
            $\mathcal{L}_\text{NPO}=\dfrac{2}{\beta|x|}\sum_i \log\!\Big(1+\big(\tfrac{P_\theta}{P_{\theta_\mathcal{D}}}\big)^{\!\beta}\Big)$
        \end{columns}}

    \vspace{1.0em}
    \uncover<3->{\small All three \textbf{push probability away} from the forget documents.}
\end{frame}

\begin{frame}{What \RESTOR\ Tells Us}
    \vspace{0.5em}
    \begin{itemize}\small\setlength\itemsep{0.7em}
        \item Forgetting and restoring are \textbf{different targets}. A benchmark that only
              checks forgetting cannot tell them apart.
        \item GA and KL forget and stop there. They push the model toward
              \textcolor{xattn}{\textbf{random}}, which reads as success on a forgetting metric.
        \item \uncover<2->{When the correct facts are \textbf{still in the training data},
                    data-level unlearning is achievable: NPO comes back to within two points
                    of the clean model.}
        \item \uncover<3->{When corruption is deep enough, \textbf{no current algorithm
                    recovers anything}. Success is contingent on what is left to recover.}
    \end{itemize}
\end{frame}

\begin{frame}{Everything Happens at the Final Checkpoint}
    \centering
    \vspace{0.6em}
    {\small Every algorithm so far reads exactly one model:
        \textcolor{themecolor}{\textbf{$\theta_\mathcal{D}$, the final checkpoint}}.
        It pushes probability down on $\mathcal{D}_f$ and does not control where it lands.}

    \vspace{0.9em}
    \uncover<2->{\small \textbf{Task arithmetic} comes at it from the other side:
        keep training $\theta_\mathcal{D}$ on $\mathcal{D}_f$, then step \textbf{backwards}.}

    \vspace{0.6em}
    \uncover<2->{\small\[
            \vec\theta_f=\theta'_\mathcal{D}-\theta_\mathcal{D},
            \qquad
            \theta_\text{unlearn}=\theta_\mathcal{D}-\alpha\,\vec\theta_f
        \]}

    \vspace{0.6em}
    \uncover<3->{\small But $\theta_\mathcal{D}$ has \textit{already} absorbed $\mathcal{D}_f$.
        The signal is \textbf{entangled} with everything the model knows, so the direction
        is a \textcolor{xattn}{\textbf{noisy, biased}} estimate of what the data did.}

    \vspace{0.7em}
    \uncover<4->{\small \textcolor{themecolor}{\textbf{What if we took the direction from
                before the data arrived?}}}
\end{frame}

\begin{frame}{Which Checkpoint, and Why It Helps}
    \vspace{0.4em}
    \begin{itemize}\small\setlength\itemsep{0.6em}
        \item \textbf{The intuition.} $\theta_C$ has never seen $\mathcal{D}_f$. Finetune it on
              $\mathcal{D}_f$ and the movement in weight space is \textbf{caused by that data alone}.
        \item \uncover<2->{\textbf{Why not $\theta_\mathcal{D}$?} It already absorbed $\mathcal{D}_f$.
                    Finetuning it again mixes the data's effect with everything else it knows.}
        \item \uncover<3->{\textbf{Which one?} Any checkpoint from before $\mathcal{D}_f$ entered
                    training. Base and instruct releases already qualify when the target arrives
                    at finetuning time.}
        \item \uncover<4->{\textbf{Why not just ship $\theta_C$?} The model learned a great deal
                    after that point. Unlearning has to keep all of it.}
    \end{itemize}
\end{frame}

\begin{frame}{How Old Can the Checkpoint Be?}
    \vspace{0.6em}
    \begin{itemize}\small\setlength\itemsep{0.7em}
        \item On \RESTOR, even the \textbf{500B}-token checkpoint is competitive:
              misinformation leaves a forget vector that is distinctive in weight space.
        \item \uncover<2->{On TOFU, the two earliest checkpoints slip on recovery and
                    retain accuracy. A checkpoint within a few hundred billion tokens
                    is enough.}
        \item \uncover<3->{On MUSE, early checkpoints \textbf{over-shoot}: the forget set
                    is gone, but so is knowledge that should have stayed.}
    \end{itemize}

    \vspace{1.0em}
    \uncover<4->{\small A gap of \textbf{hundreds of billions of tokens} is survivable.
        \textcolor{xattn}{\textbf{Trillions are not.}}}
\end{frame}

\begin{frame}{What the Two Papers Say Together}
    \centering
    \vspace{0.8em}
    \begin{itemize}\small
        \item Concept-level benchmarks reward \textbf{silence}. \RESTOR\ asks for \textbf{recovery}.
        \item Loss-maximizing methods forget, then drift to random.
        \item No single method wins on every metric.
        \item \textbf{Training history is a resource}: a checkpoint from before the data
              gives a cleaner estimate of what that data did.
    \end{itemize}
\end{frame}

\begin{frame}{How the Winner Is Picked}
    \centering
    \vspace{0.4em}
    \begin{tcolorbox}[colback=themecolor!4,colframe=themecolor,arc=1.5mm,boxrule=0.9pt,
            width=0.88\textwidth,left=8pt,right=8pt,top=5pt,bottom=5pt,
            fonttitle=\bfseries\small,coltitle=white,colbacktitle=themecolor,
            title=Segment auction]
        \small
        For each segment:\\[0.4em]
        \hspace*{1.2em} score $\;s_i = \underbrace{q_i}_{\text{relevance}}\cdot
        \underbrace{b_i}_{\text{bid}}\cdot\; e^{\varepsilon_i}$,
        \quad $\varepsilon_i \sim \text{Gumbel}$\\[0.4em]
        \hspace*{1.2em} the winner is $\arg\max_i s_i$\\[0.4em]
        \hspace*{1.2em} it pays the \textbf{lowest bid that would still have won},
        charged \textbf{per click}
    \end{tcolorbox}

    \vspace{0.8em}
    \uncover<2->{\small The noise makes it a \textbf{lottery weighted by $q_i b_i$}:
        a highly relevant ad can beat a higher bid.}

    \vspace{0.5em}
    \uncover<3->{\small The price is the second-price idea, so
        \textcolor{themecolor}{\textbf{bidding your true value is a dominant strategy}}.}
\end{frame}

\begin{frame}{What the Mechanism Guarantees}
    \vspace{0.5em}
    \begin{itemize}\small\setlength\itemsep{0.7em}
        \item \textbf{Truthful} (DSIC). No advertiser gains by misreporting its value.
        \item \textbf{Individually rational}. Nobody is worse off for taking part.
        \item \uncover<2->{\textbf{Maximal logarithmic social welfare}
                  $\prod_i x_i^{\,v_i q_i}$ among independent segment auctions.}
        \item \uncover<3->{A \textbf{log} welfare, not a sum. It rewards spreading
                  allocation across advertisers rather than handing every segment to one,
                  so efficiency and fairness come together.}
    \end{itemize}

    \vspace{0.9em}
    \uncover<4->{\small Ignoring relevance earns more revenue and serves the user worse.}

\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion}
    \vspace{0.5em}
    {\small This thesis asks where a foundation model's knowledge comes from,
        and what we can do once we know.}

    \vspace{0.9em}
    \begin{itemize}\small\setlength\itemsep{0.7em}
        \item \textbf{Where knowledge lives.} Methods and heuristics for localizing a
              concept inside a generative model, and for editing it in place.
        \item \uncover<2->{\textbf{What the data did.} Tools for studying the effect of
                    training data through machine unlearning: a benchmark that asks for
                    recovery, and a method that reads the training history.}
        \item \uncover<3->{\textbf{How it sustains itself.} Mechanism design for
                    generative AI: auctions that place advertising inside a model's output,
                    with the incentive guarantees such a market needs.}
    \end{itemize}
